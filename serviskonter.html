Panduan Kode Google Apps Script (Sheet: APLIKASI_KONTER)

Kode ini mendukung sheet kosong & akan otomatis membuat header kolom jika belum tersedia.

// Google Apps Script (Backend for Sheet: APLIKASI_KONTER)
// Tempelkan kode ini di Ekstensi > Apps Script pada Spreadsheet Anda.

const SHEET_NAME = "APLIKASI_KONTER";
const HEADERS = [
  "ID_Pendaftaran",
  "Nama_HP_Petugas",
  "Detail_Kerusakan",
  "Nama_Konsumen",
  "Jenis_Kelamin",
  "No_HP_Konsumen",
  "Detail_Alamat",
  "Latitude",
  "Longitude",
  "Status_Perbaikan",
  "Biaya_Perbaikan",
  "Tanggal_Daftar"
];

function getOrCreateSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
  }
  // Buat header otomatis jika sheet masih kosong
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(HEADERS);
    var headerRange = sheet.getRange(1, 1, 1, HEADERS.length);
    headerRange.setFontWeight("bold");
    headerRange.setBackground("#0284c7");
    headerRange.setFontColor("#ffffff");
  }
  return sheet;
}

function doGet(e) {
  try {
    var sheet = getOrCreateSheet();
    var data = sheet.getDataRange().getValues();
    if (data.length <= 1) {
      return ContentService.createTextOutput(JSON.stringify([]))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    var headers = data[0];
    var rows = data.slice(1);
    
    var result = rows.map(function(row) {
      var obj = {};
      headers.forEach(function(header, index) {
        obj[header] = row[index];
      });
      return obj;
    });
    
    return ContentService.createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "ERROR", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    var sheet = getOrCreateSheet();
    var contents = e.postData.contents;
    var data = JSON.parse(contents);
    
    var values = sheet.getDataRange().getValues();
    var rowIndex = -1;
    for (var i = 1; i < values.length; i++) {
      if (values[i][0] == data.ID_Pendaftaran) {
        rowIndex = i + 1; // 1-based row index
        break;
      }
    }
    
    if (data.action === "UPDATE" && rowIndex > -1) {
      sheet.getRange(rowIndex, 3).setValue(data.Detail_Kerusakan);
      sheet.getRange(rowIndex, 10).setValue(data.Status_Perbaikan);
      sheet.getRange(rowIndex, 11).setValue(data.Biaya_Perbaikan);
      return ContentService.createTextOutput(JSON.stringify({ status: "SUCCESS", message: "Data updated" }))
        .setMimeType(ContentService.MimeType.JSON);
    } else if (data.action === "DELETE" && rowIndex > -1) {
      sheet.deleteRow(rowIndex);
      return ContentService.createTextOutput(JSON.stringify({ status: "SUCCESS", message: "Data deleted" }))
        .setMimeType(ContentService.MimeType.JSON);
    } else {
      // ADD / UPSERT
      var rowData = [
        data.ID_Pendaftaran,
        data.Nama_HP_Petugas,
        data.Detail_Kerusakan,
        data.Nama_Konsumen,
        data.Jenis_Kelamin,
        data.No_HP_Konsumen,
        data.Detail_Alamat,
        data.Latitude,
        data.Longitude,
        data.Status_Perbaikan,
        data.Biaya_Perbaikan,
        data.Tanggal_Daftar || new Date().toLocaleString("id-ID")
      ];
      if (rowIndex > -1) {
        sheet.getRange(rowIndex, 1, 1, rowData.length).setValues([rowData]);
      } else {
        sheet.appendRow(rowData);
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "SUCCESS", message: "Data saved" }))
        .setMimeType(ContentService.MimeType.JSON);
    }
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "ERROR", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
