Script Google Apps Script Terbaru

Salin kode di bawah ini ke editor Google Apps Script spreadsheet Anda untuk otomatis merekam tanggal, waktu, NIS, NISN, nama, gender, kelas, dan status kehadiran: 

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    // Buat header otomatis jika baris 1 masih kosong
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Tanggal", "Waktu", "NIS", "NISN", "Nama Siswa", "Gender", "Kelas", "Status"]);
    }
    
    var data = JSON.parse(e.postData.contents);
    
    var tanggal = data.tanggal || "";
    var waktu = data.waktu || "";
    var nis = data.nis || "";
    var nisn = data.nisn || "";
    var nama = data.nama || "";
    var gender = data.gender || "";
    var kelas = data.kelas || "";
    var status = data.status || "";
    
    // Cek apakah siswa sudah absen di tanggal yang sama, jika ada update barisnya atau buat baru
    var rows = sheet.getDataRange().getValues();
    var updated = false;
    
    for (var i = 1; i < rows.length; i++) {
      if (rows[i][0] === tanggal && rows[i][2] === nis) {
        sheet.getRange(i + 1, 2).setValue(waktu);
        sheet.getRange(i + 1, 8).setValue(status);
        updated = true;
        break;
      }
    }
    
    if (!updated) {
      sheet.appendRow([tanggal, waktu, nis, nisn, nama, gender, kelas, status]);
    }
    
    return ContentService.createTextOutput(JSON.stringify({"result": "success"})).setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "message": error.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
